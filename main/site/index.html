main.py — قدرت LazyNoto 
from lazynoto import LazyNoto, NotoContext, InlineMenu

bot = LazyNoto(token="YOUR_TELEGRAM_BOT_TOKEN")

# دکمه‌های شیشه‌ای شیک به روش زنجیره‌ای
menu = (
    InlineMenu()
    .button("🛒 لیست پلن‌ها", callback_data="plans")
    .button("👤 پروفایل کاربری", callback_data="profile")
    .build()
)

@bot.command("start")
async def start(ctx: NotoContext):
    await ctx.reply(
        f"سلام {ctx.user.first_name} عزیز! 🌟\nبه پنل هوشمند خوش آمدید.",
        reply_markup=menu,
        parse_mode="HTML"
    )

@bot.callback("plans")
async def show_plans(ctx: NotoContext):
    await ctx.answer_callback("در حال لود سرویس‌ها...")
    await ctx.edit("⚡ پلن حرفه‌ای ۳ ماهه فعال شد!")

if __name__ == "__main__":
    bot.run()
